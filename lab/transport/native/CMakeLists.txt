# Maffinet P01. Mirrors pinned upstream Android.mk/build.mk/configs.mk sources
# and definitions; CMake quoted paths support the workspace's spaces.
cmake_minimum_required(VERSION 3.22.1)
project(maffinet_transport_lab C ASM)
if(NOT ANDROID)
    message(FATAL_ERROR "Use the provisioned Android NDK; this is not a host TUN fixture")
endif()
get_filename_component(WORKSPACE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT HEV_SOURCE)
    set(HEV_SOURCE "${WORKSPACE}/.toolchain/transport-source")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${WORKSPACE}/tools/verify-transport-source.py" "${HEV_SOURCE}"
    RESULT_VARIABLE SOURCE_CHECK)
if(NOT SOURCE_CHECK EQUAL 0)
    message(FATAL_ERROR "Pinned/patch source verification failed; prepare a fresh output")
endif()
set(SRC "${HEV_SOURCE}/src")
set(TASK "${HEV_SOURCE}/third-part/hev-task-system")
set(LWIP "${HEV_SOURCE}/third-part/lwip")
set(YAML "${HEV_SOURCE}/third-part/yaml")
file(GLOB_RECURSE TASK_FILES CONFIGURE_DEPENDS "${TASK}/src/*.c" "${TASK}/src/*.S")
file(GLOB_RECURSE LWIP_FILES CONFIGURE_DEPENDS "${LWIP}/src/*.c" "${LWIP}/src/*.S")
file(GLOB_RECURSE YAML_FILES CONFIGURE_DEPENDS "${YAML}/src/*.c" "${YAML}/src/*.S")
file(GLOB_RECURSE HEV_FILES CONFIGURE_DEPENDS "${SRC}/*.c" "${SRC}/*.S")
list(REMOVE_ITEM HEV_FILES "${SRC}/hev-main.c" "${SRC}/hev-jni.c")
add_library(lab-task STATIC ${TASK_FILES})
target_include_directories(lab-task PUBLIC "${TASK}/include" PRIVATE "${TASK}/src")
target_compile_definitions(lab-task PRIVATE ENABLE_STACK_OVERFLOW_DETECTION ENABLE_MEMALLOC_SLICE ENABLE_IO_SPLICE_SYSCALL
    CONFIG_STACK_BACKEND=STACK_MMAP CONFIG_STACK_OVERFLOW_DETECTION=1 CONFIG_MEMALLOC_SLICE_ALIGN=64
    CONFIG_MEMALLOC_SLICE_MAX_SIZE=4096 CONFIG_MEMALLOC_SLICE_MAX_COUNT=1000 CONFIG_SCHED_CLOCK=CLOCK_NONE)
target_compile_options(lab-task PRIVATE -fvisibility=hidden)
add_library(lab-lwip STATIC ${LWIP_FILES})
target_include_directories(lab-lwip PUBLIC "${LWIP}/src/include" "${LWIP}/src/ports/include")
target_compile_definitions(lab-lwip PRIVATE FD_SET_DEFINED SOCKLEN_T_DEFINED)
add_library(lab-yaml STATIC ${YAML_FILES})
target_include_directories(lab-yaml PUBLIC "${YAML}/src")
target_compile_definitions(lab-yaml PRIVATE YAML_VERSION_MAJOR=0 YAML_VERSION_MINOR=2 YAML_VERSION_PATCH=5 YAML_VERSION_STRING="0.2.5")
add_library(maffinet-transport-lab SHARED ${HEV_FILES} bridge.c main.c)
target_include_directories(maffinet-transport-lab PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${SRC}" "${SRC}/misc" "${SRC}/core/src")
target_compile_definitions(maffinet-transport-lab PRIVATE FD_SET_DEFINED SOCKLEN_T_DEFINED ENABLE_LIBRARY COMMIT_ID="4d6c334")
target_link_libraries(maffinet-transport-lab PRIVATE lab-yaml lab-lwip lab-task)
foreach(target IN ITEMS lab-task lab-lwip lab-yaml maffinet-transport-lab)
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(ANDROID_ABI STREQUAL "armeabi-v7a")
        target_compile_options(${target} PRIVATE -mfpu=neon)
    endif()
endforeach()
target_link_options(maffinet-transport-lab PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
